# Exceptions for scripts/check-config-sync.sh — every line carries its reason.
#
# Three entry kinds:
#   triple <json.path> <ENV_NAME>   a real config read the extractor regex
#                                   cannot see (it only parses literal
#                                   cfg.get<T>("path", "ENV", ...) calls and
#                                   cfg.get(Keys::name, ...) registry reads)
#   absent-json <json.path>         a key deliberately NOT present in
#                                   config/config.json + config.sample.json
#   defaults-differ <json.path> <ENV_NAME>
#                                   two sites read the knob with two different
#                                   literal defaults on purpose, both documented
#
# Rot protection is built into the gate: an entry that becomes satisfiable
# (triple now parsed directly, absent-json key added to the jsons or no
# longer read) FAILS the gate until the entry is deleted.

# Core.cpp passes these key/env pairs as const char* arguments into the
# shared register_depth_metric_() helper, which performs the cfg.get<int>()
# — one call site, two knobs, invisible to the literal-triple regex.
triple jobs.dlq_metric_refresh_sec JOBS_DLQ_METRIC_REFRESH_SEC
triple jobs.queue_metric_refresh_sec JOBS_QUEUE_METRIC_REFRESH_SEC

# The code default is Utils::Strings::kDefaultProtectedPathsCsv (the login/
# register/refresh brute-force surface). AppConfig::get<std::string> returns
# a present-but-empty string AS-IS (only non-string types fall back to the
# default on ""), so a '${RATE_LIMIT_PROTECTED_PATHS:-}' placeholder in the
# json would silently EMPTY the protected set — and duplicating the C++
# constant into the json would recreate the api.public_paths full-override
# footgun. Env-or-code-default only, on purpose.
absent-json rate_limit.protected_paths

# Core::init_database_ defaults database.primary to "" and assembles the DSN
# from the DATABASE_* parts; `--verify-migrations` (main.cpp) runs without
# Core and falls back to postgresql://localhost:5432/app. docs/CONFIG.md
# documents both in the DATABASE_PRIMARY_URL row.
defaults-differ database.primary DATABASE_PRIMARY_URL

# compose-absent: envs the helm deployments set that docker/docker-compose.yml
# deliberately leaves to config/config.json (or worker.json), because they are
# Kubernetes plumbing or prod tuning with no meaning for `make up`.
# DATABASE_PASSWORD: the chart splits the password out of the URL into a
# Secret; compose carries it inside DATABASE_PRIMARY_URL.
compose-absent app DATABASE_PASSWORD
compose-absent worker DATABASE_PASSWORD
# DB_MIGRATIONS_ENABLED / RUN_MIGRATIONS_ONLY: the chart runs migrations in a
# migrate init container and turns them off in the main one; compose lets the
# app run them at start, the config json's default.
compose-absent app DB_MIGRATIONS_ENABLED
compose-absent worker DB_MIGRATIONS_ENABLED
compose-absent app RUN_MIGRATIONS_ONLY
# SHUTDOWN_PRE_STOP_DELAY_SEC: the delay that lets a pod leave the Service
# endpoints before it stops; compose has no endpoints.
compose-absent app SHUTDOWN_PRE_STOP_DELAY_SEC
# DB_POOL_SIZE, CACHE_POOL_SIZE: prod tuning from chart values; the config
# json's values fit a laptop.
compose-absent app DB_POOL_SIZE
compose-absent worker DB_POOL_SIZE
compose-absent app CACHE_POOL_SIZE
compose-absent worker CACHE_POOL_SIZE
# DB_STATEMENT_TIMEOUT_MS, METRICS_ADDRESS: set only on the one-shot init
# containers (no statement timeout for the migrate step, no metrics exposer
# on a process that exits); the long-running containers use the json's values.
compose-absent app DB_STATEMENT_TIMEOUT_MS
compose-absent app METRICS_ADDRESS
compose-absent worker METRICS_ADDRESS
# APP_ENV, LOG_FORMAT, LOG_NAME: the chart sets production, its log format
# and a per-chart log name; compose keeps the config json's dev values
# (development, text logs).
compose-absent app APP_ENV
compose-absent app LOG_FORMAT
compose-absent worker LOG_FORMAT
compose-absent app LOG_NAME
compose-absent worker LOG_NAME
